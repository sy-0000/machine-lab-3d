const MODEL='lathe_16k20_dark_green_no_backboard.glb';
const snap=page=>page.evaluate(()=>window.__LATHE_DEBUG__);
async function open(page){await page.setViewportSize({width:1440,height:1900});await page.goto('/?inspect=1');await expect(page.getByRole('button',{name:'▶ 啟動主軸'})).toBeEnabled({timeout:30000});await page.waitForFunction(()=>window.__LATHE_DEBUG__?.inputReady);}
async function point(page,key){await expect.poll(async()=>(await snap(page)).picks[key]?.length||0).toBeGreaterThan(0);return(await snap(page)).picks[key][0];}
async function reset(page){await page.getByRole('button',{name:'↺ 重設操作與視角'}).click();await expect.poll(async()=>Object.values((await snap(page)).angles).every(x=>x===0)).toBe(true);}

test('all four wheels: left negative, right positive, linked slider, camera fixed and immediate release',async({page})=>{
 await open(page);
 for(const [key,axis]of [['carriageHandwheel','x'],['crossSlideHandwheel','y'],['heightHandwheel','z'],['tailstockHandwheel','quill']]){
  await reset(page);let p=await point(page,key);await page.mouse.move(p.x,p.y);await page.mouse.down({button:'left'});
  await expect.poll(async()=>(await snap(page)).orbitEnabled).toBe(false);const camera=(await snap(page)).camera;
  await expect.poll(async()=>(await snap(page)).angles[key]).toBeLessThan(-0.1);
  // Pointer movement must not change the rotation direction.
  await page.mouse.move(p.x+15,p.y+15);await page.mouse.up({button:'left'});
  await expect.poll(async()=>(await snap(page)).interaction.pressed).toBe(false);const left=(await snap(page)).angles[key];expect(left).toBeLessThan(0);expect(Number(await page.locator(`#${axis}`).inputValue())).toBeLessThan(0);
  expect((await snap(page)).camera).toEqual(camera);await page.waitForTimeout(150);expect((await snap(page)).angles[key]).toBe(left);
  p=await point(page,key);await page.mouse.move(p.x,p.y);await page.mouse.down({button:'right'});await expect.poll(async()=>(await snap(page)).angles[key]).toBeGreaterThan(left);await page.mouse.up({button:'right'});await expect.poll(async()=>(await snap(page)).orbitEnabled).toBe(true);
 }
});
test('context menu suppressed only over wheels; cancel, lost capture, blur and reset restore camera',async({page})=>{
 await open(page);let p=await point(page,'carriageHandwheel');
 const cancelled=await page.locator('canvas').evaluate((canvas,p)=>{const event=new MouseEvent('contextmenu',{clientX:p.x,clientY:p.y,bubbles:true,cancelable:true});canvas.dispatchEvent(event);return event.defaultPrevented;},p);expect(cancelled).toBe(true);
 const normal=await page.locator('canvas').evaluate(canvas=>{const r=canvas.getBoundingClientRect();const event=new MouseEvent('contextmenu',{clientX:r.left+10,clientY:r.top+100,bubbles:true,cancelable:true});canvas.dispatchEvent(event);return event.defaultPrevented;});expect(normal).toBe(false);
 expect(await page.locator('header').evaluate(el=>{const e=new MouseEvent('contextmenu',{bubbles:true,cancelable:true});el.dispatchEvent(e);return e.defaultPrevented;})).toBe(false);
 for(const type of ['pointercancel','lostpointercapture','blur','reset']){
  p=await point(page,'carriageHandwheel');await page.mouse.move(p.x,p.y);await page.mouse.down();await expect.poll(async()=>(await snap(page)).orbitEnabled).toBe(false);
  if(type==='blur')await page.evaluate(()=>window.dispatchEvent(new Event('blur')));
  else if(type==='reset'){await page.getByRole('button',{name:'↺ 重設操作與視角'}).focus();await page.keyboard.press('Enter');}
  else await page.locator('canvas').dispatchEvent(type,{pointerId:1,pointerType:'mouse'});
  await expect.poll(async()=>(await snap(page)).interaction.pressed).toBe(false);await expect.poll(async()=>(await snap(page)).orbitEnabled).toBe(true);await page.mouse.up();
 }
});
